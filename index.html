function carregar(id){ let likes = Number(localStorage.getItem("likes"+id)) || 0; let dislikes = Number(localStorage.getItem("dislikes"+id)) || 0; document.getElementById("likes"+id).textContent = likes; document.getElementById("dislikes"+id).textContent = dislikes; let voto = localStorage.getItem("voto"+id); const likeBtn = document.getElementById("likeBtn"+id); const dislikeBtn = document.getElementById("dislikeBtn"+id); likeBtn.style.opacity = "1"; dislikeBtn.style.opacity = "1"; if(voto == "like"){ likeBtn.style.background = "#1e7e34"; } if(voto == "dislike"){ dislikeBtn.style.background = "#a71d2a"; } } function votar(id,tipo){ let likes = Number(localStorage.getItem("likes"+id)) || 0; let dislikes = Number(localStorage.getItem("dislikes"+id)) || 0; let votoAnterior = localStorage.getItem("voto"+id); // Mesmo voto if(votoAnterior == tipo){ return; } // Remove voto antigo if(votoAnterior == "like"){ likes--; } if(votoAnterior == "dislike"){ dislikes--; } // Adiciona novo voto if(tipo == "like"){ likes++; }else{ dislikes++; } localStorage.setItem("likes"+id, likes); localStorage.setItem("dislikes"+id, dislikes); localStorage.setItem("voto"+id, tipo); carregar(id); } carregar(1); carregar(2);
